// ════════════════════════════════════════════════════════════
// ═══ NOVO ═══  WI-FI — CONEXÃO
// ════════════════════════════════════════════════════════════

void conectarWiFi()
{
  if (WiFi.status() == WL_CONNECTED)
  {
    wifiConectado = true;
    return;
  }

  wifiConectado = false;

  Serial.print("Conectando Wi-Fi");
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASS);

  unsigned long inicio = millis();

  while (WiFi.status() != WL_CONNECTED && millis() - inicio < 10000)
  {
    delay(500);
    Serial.print(".");
  }

  Serial.println();

  if (WiFi.status() == WL_CONNECTED)
  {
    wifiConectado = true;
    Serial.print("Wi-Fi OK. IP: ");
    Serial.println(WiFi.localIP());
  }
  else
  {
    Serial.println("Falha ao conectar Wi-Fi. Tentara de novo.");
  }
}


// ════════════════════════════════════════════════════════════
// ═══ NOVO ═══  CONVERSÃO DE ESTADO PARA STRING (JSON)
// ════════════════════════════════════════════════════════════

const char* estadoParaString(EstadoAlagamento e)
{
  switch (e)
  {
    case NORMAL:       return "normal";
    case ATENCAO:      return "attention";
    case ALERTA:       return "alert";
    case PERIGO:       return "danger";
    case CRITICO:      return "critical";
    case SENSOR_FALHA: return "sensor_fault";
  }
  return "normal";
}

const char* tendenciaParaString()
{
  if (subidaRapida)      return "rising_fast";
  if (taxaSubida > 0.3)  return "rising";
  if (taxaSubida < -0.3) return "falling";
  return "stable";
}


// ════════════════════════════════════════════════════════════
// ═══ NOVO ═══  ENVIO DE TELEMETRIA PARA A NUVEM
// ════════════════════════════════════════════════════════════

void enviarTelemetria()
{
  // --------------------------------------------------------
  // Garante Wi-Fi
  // --------------------------------------------------------

  if (WiFi.status() != WL_CONNECTED)
  {
    conectarWiFi();

    if (WiFi.status() != WL_CONNECTED)
    {
      totalFalhas++;
      return;
    }
  }


  // --------------------------------------------------------
  // Monta o JSON
  // --------------------------------------------------------

  StaticJsonDocument<512> doc;

  doc["distance_cm"]      = distanciaAtual;
  doc["water_depth_cm"]   = aguaAtual;
  doc["rise_rate_cm_min"] = taxaSubida;
  doc["alert"]            = estadoParaString(estadoAtual);
  doc["trend"]            = tendenciaParaString();
  doc["sensor_ok"]        = sensorOK;
  doc["wifi_rssi"]        = WiFi.RSSI();
  doc["firmware_version"] = FIRMWARE_VERSION;

  String body;
  serializeJson(doc, body);


  // --------------------------------------------------------
  // POST
  // --------------------------------------------------------

  HTTPClient http;
  http.begin(API_URL);
  http.addHeader("Content-Type", "application/json");
  http.addHeader("Authorization", String("Bearer ") + DEVICE_TOKEN);
  http.setTimeout(5000);

  int code = http.POST(body);

  if (code == 200)
  {
    totalEnvios++;

    Serial.print("POST 200 | Envios: ");
    Serial.print(totalEnvios);
    Serial.print(" | Falhas: ");
    Serial.println(totalFalhas);
  }
  else
  {
    totalFalhas++;

    Serial.print("POST FALHOU | code=");
    Serial.println(code);
  }

  http.end();
}


// ============================================================
// MEDIÇÃO HC-SR04
// ============================================================

float medirDistancia()
{
  digitalWrite(TRIG, LOW);
  delayMicroseconds(3);

  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);

  digitalWrite(TRIG, LOW);

  unsigned long duracao = pulseIn(ECHO, HIGH, 30000);

  if (duracao == 0)
  {
    return -1.0;
  }

  float distancia = (duracao * 0.0343) / 2.0;

  // Faixa aceitável
  if (distancia < 2.0 || distancia > 450.0)
  {
    return -1.0;
  }

  return distancia;
}


// ============================================================
// MEDIANA
// ============================================================

float calcularMediana(float valores[], int tamanho)
{
  float copia[TAMANHO_FILTRO];

  for (int i = 0; i < tamanho; i++)
  {
    copia[i] = valores[i];
  }

  for (int i = 0; i < tamanho - 1; i++)
  {
    for (int j = i + 1; j < tamanho; j++)
    {
      if (copia[j] < copia[i])
      {
        float temp = copia[i];
        copia[i] = copia[j];
        copia[j] = temp;
      }
    }
  }

  return copia[tamanho / 2];
}


// ============================================================
// FILTRO
// ============================================================

float filtrarDistancia(float novaDistancia)
{
  if (!filtroInicializado)
  {
    for (int i = 0; i < TAMANHO_FILTRO; i++)
    {
      historicoDistancia[i] = novaDistancia;
    }

    indiceFiltro = 0;
    filtroInicializado = true;

    return novaDistancia;
  }

  historicoDistancia[indiceFiltro] = novaDistancia;

  indiceFiltro++;

  if (indiceFiltro >= TAMANHO_FILTRO)
  {
    indiceFiltro = 0;
  }

  return calcularMediana(
    historicoDistancia,
    TAMANHO_FILTRO
  );
}


// ============================================================
// CALCULA ÁGUA
// ============================================================

float calcularAgua(float distancia)
{
  float agua = DISTANCIA_PISO_CM - distancia;

  if (agua < 0.0)
  {
    agua = 0.0;
  }

  return agua;
}


// ============================================================
// DETERMINA ESTADO DIRETO PELA ÁGUA
// ============================================================
//
// IMPORTANTE:
// Não existe mais progressão artificial.
//
// 25 cm de água = CRÍTICO imediatamente.
// 13 cm = PERIGO imediatamente.
// ============================================================

EstadoAlagamento determinarEstadoDireto(float agua)
{
  if (agua >= LIMITE_CRITICO)
  {
    return CRITICO;
  }

  if (agua >= LIMITE_PERIGO)
  {
    return PERIGO;
  }

  if (agua >= LIMITE_ALERTA)
  {
    return ALERTA;
  }

  if (agua >= LIMITE_ATENCAO)
  {
    return ATENCAO;
  }

  return NORMAL;
}


// ============================================================
// HISTERESE
// ============================================================

EstadoAlagamento aplicarHisterese(float agua)
{
  switch (estadoAtual)
  {
    case NORMAL:

      if (agua >= LIMITE_ATENCAO)
        return determinarEstadoDireto(agua);

      return NORMAL;


    case ATENCAO:

      if (agua >= LIMITE_ATENCAO + HISTERESE)
        return determinarEstadoDireto(agua);

      if (agua < LIMITE_ATENCAO - HISTERESE)
        return NORMAL;

      return ATENCAO;


    case ALERTA:

      if (agua >= LIMITE_ALERTA + HISTERESE)
        return determinarEstadoDireto(agua);

      if (agua < LIMITE_ALERTA - HISTERESE)
        return determinarEstadoDireto(agua);

      return ALERTA;


    case PERIGO:

      if (agua >= LIMITE_PERIGO + HISTERESE)
        return determinarEstadoDireto(agua);

      if (agua < LIMITE_PERIGO - HISTERESE)
        return determinarEstadoDireto(agua);

      return PERIGO;


    case CRITICO:

      if (agua < LIMITE_CRITICO - HISTERESE)
        return determinarEstadoDireto(agua);

      return CRITICO;


    case SENSOR_FALHA:

      return determinarEstadoDireto(agua);
  }

  return NORMAL;
}


// ============================================================
// ATUALIZA ESTADO
// ============================================================

void atualizarEstado()
{
  if (!sensorOK)
  {
    if (estadoAtual != SENSOR_FALHA)
    {
      estadoAtual = SENSOR_FALHA;
      estadoCandidato = SENSOR_FALHA;
      contadorConfirmacaoEstado = 0;

      Serial.println();
      Serial.println("==========================================");
      Serial.println("NOVO ESTADO: SENSOR_FALHA");
      Serial.println("==========================================");
    }

    return;
  }


  EstadoAlagamento desejado = aplicarHisterese(aguaAtual);


  // ==========================================================
  // SUBIDA RÁPIDA
  // ==========================================================
  //
  // Pode elevar no máximo UM nível.
  //
  // Isso evita:
  // NORMAL -> CRITICO
  //
  // apenas por uma leitura instantânea.
  // ==========================================================

  if (subidaRapida && aguaAtual >= 2.0)
  {
    if (desejado == NORMAL)
      desejado = ATENCAO;

    else if (desejado == ATENCAO)
      desejado = ALERTA;

    else if (desejado == ALERTA)
      desejado = PERIGO;

    else if (desejado == PERIGO)
      desejado = CRITICO;
  }


  // ==========================================================
  // SE FOR IGUAL AO ESTADO ATUAL
  // ==========================================================

  if (desejado == estadoAtual)
  {
    estadoCandidato = estadoAtual;
    contadorConfirmacaoEstado = 0;
    return;
  }


  // ==========================================================
  // NOVO CANDIDATO
  // ==========================================================

  if (desejado != estadoCandidato)
  {
    estadoCandidato = desejado;
    contadorConfirmacaoEstado = 1;
    return;
  }


  // ==========================================================
  // CONFIRMAÇÃO
  // ==========================================================

  contadorConfirmacaoEstado++;

  if (contadorConfirmacaoEstado >= LEITURAS_PARA_CONFIRMAR)
  {
    estadoAtual = desejado;

    contadorConfirmacaoEstado = 0;

    Serial.println();
    Serial.println("==========================================");
    Serial.print("NOVO ESTADO: ");
    Serial.println(nomeEstado(estadoAtual));
    Serial.println("==========================================");
  }
}


// ============================================================
// ATUALIZA SENSOR
// ============================================================

void atualizarSensor()
{
  float leitura = medirDistancia();


  // ==========================================================
  // LEITURA INVÁLIDA
  // ==========================================================

  if (leitura < 0)
  {
    falhasSensor++;

    recuperacoesSensor = 0;

    if (falhasSensor >= FALHAS_SENSOR)
    {
      sensorOK = false;
    }

    return;
  }


  // ==========================================================
  // LEITURA VÁLIDA
  // ==========================================================

  falhasSensor = 0;

  recuperacoesSensor++;


  // ==========================================================
  // RECUPERAÇÃO DO SENSOR
  // ==========================================================

  if (!sensorOK)
  {
    if (recuperacoesSensor >= RECUPERACOES_SENSOR)
    {
      sensorOK = true;

      Serial.println();
      Serial.println("SENSOR RECUPERADO.");
      Serial.println();
    }
    else
    {
      return;
    }
  }


  // ==========================================================
  // FILTRO
  // ==========================================================

  distanciaFiltrada = filtrarDistancia(leitura);

  distanciaAtual = distanciaFiltrada;


  // ==========================================================
  // ÁGUA
  // ==========================================================

  float novaAgua = calcularAgua(distanciaAtual);

  aguaAnterior = aguaAtual;

  aguaAtual = novaAgua;


  // ==========================================================
  // ATUALIZA TAXA
  // ==========================================================

  unsigned long agora = millis();

  if (ultimaTaxa == 0)
  {
    ultimaTaxa = agora;

    aguaAnterior = aguaAtual;

    taxaSubida = 0.0;
    subidaRapida = false;
  }
  else if (agora - ultimaTaxa >= INTERVALO_TAXA)
  {
    float deltaAgua = aguaAtual - aguaAnterior;

    float tempoMinutos =
      (agora - ultimaTaxa) / 60000.0;


    if (tempoMinutos > 0.0)
    {
      float taxaNova = deltaAgua / tempoMinutos;


      // ======================================================
      // PROTEÇÃO CONTRA SALTO ABSURDO
      // ======================================================

      if (taxaNova > 100.0)
      {
        taxaNova = 100.0;
      }

      if (taxaNova < -100.0)
      {
        taxaNova = -100.0;
      }


      // ======================================================
      // FILTRO DA TAXA
      // ======================================================

      taxaSubida =
        (taxaSubida * 0.65) +
        (taxaNova * 0.35);


      if (taxaSubida >= LIMITE_SUBIDA_RAPIDA)
      {
        subidaRapida = true;
      }
      else
      {
        subidaRapida = false;
      }
    }


    aguaAnterior = aguaAtual;
    ultimaTaxa = agora;
  }
}


// ============================================================
// CONTROLE DOS BUZZERS
// ============================================================

void desligarBuzzers()
{
  noTone(BUZZER_1);
  noTone(BUZZER_2);
}


// ============================================================
// CONTROLE DOS LEDS
// ============================================================

void desligarLEDs()
{
  digitalWrite(LED_VERMELHO, LOW);
  digitalWrite(LED_AMARELO, LOW);
  digitalWrite(LED_VERDE, LOW);
}


// ============================================================
// ALARME
// ============================================================

void atualizarAlarme()
{
  unsigned long agora = millis();

  // ----------------------------------------------------------
  // SILENCIAMENTO
  // ----------------------------------------------------------

  if (buzzerSilenciado)
  {
    if (agora - inicioSilenciamento >= TEMPO_SILENCIO)
    {
      buzzerSilenciado = false;

      Serial.println("Silenciamento encerrado.");
    }
  }


  // ==========================================================
  // NORMAL
  // ==========================================================

  if (estadoAtual == NORMAL)
  {
    desligarLEDs();
    desligarBuzzers();

    digitalWrite(LED_VERDE, HIGH);

    return;
  }


  // ==========================================================
  // SENSOR FALHA
  // ==========================================================

  if (estadoAtual == SENSOR_FALHA)
  {
    desligarLEDs();

    desligarBuzzers();

    // Pisca vermelho em padrão de diagnóstico
    unsigned long ciclo = agora % 1600;

    if (ciclo < 150 || (ciclo >= 300 && ciclo < 450))
    {
      digitalWrite(LED_VERMELHO, HIGH);
    }
    else
    {
      digitalWrite(LED_VERMELHO, LOW);
    }


    if (!buzzerSilenciado)
    {
      if (ciclo < 500)
      {
        tone(BUZZER_1, 1300);
        tone(BUZZER_2, 1300);
      }
      else
      {
        desligarBuzzers();
      }
    }

    return;
  }


  // ==========================================================
  // ATENÇÃO
  // ==========================================================

  if (estadoAtual == ATENCAO)
  {
    desligarLEDs();
    desligarBuzzers();

    unsigned long ciclo = agora % 1400;

    if (ciclo < 600)
    {
      digitalWrite(LED_AMARELO, HIGH);
    }
    else
    {
      digitalWrite(LED_AMARELO, LOW);
    }

    return;
  }


  // ==========================================================
  // ALERTA
  // ==========================================================

  if (estadoAtual == ALERTA)
  {
    desligarLEDs();

    unsigned long intervalo = subidaRapida ? 250 : 500;

    unsigned long ciclo = agora % (intervalo * 2);

    if (ciclo < intervalo)
    {
      digitalWrite(LED_AMARELO, HIGH);
    }
    else
    {
      digitalWrite(LED_AMARELO, LOW);
    }


    if (!buzzerSilenciado)
    {
      unsigned long som = agora % 1200;

      if (som < 300)
      {
        tone(BUZZER_1, 1800);
      }
      else
      {
        noTone(BUZZER_1);
      }

      noTone(BUZZER_2);
    }
    else
    {
      desligarBuzzers();
    }

    return;
  }


  // ==========================================================
  // PERIGO
  // ==========================================================

  if (estadoAtual == PERIGO)
  {
    desligarLEDs();

    unsigned long intervalo = subidaRapida ? 220 : 400;

    unsigned long ciclo = agora % (intervalo * 2);

    if (ciclo < intervalo)
    {
      digitalWrite(LED_VERMELHO, HIGH);
    }
    else
    {
      digitalWrite(LED_VERMELHO, LOW);
    }


    if (!buzzerSilenciado)
    {
      unsigned long som = agora % 1600;

      if (som < 300)
      {
        tone(BUZZER_1, 2000);
        noTone(BUZZER_2);
      }
      else if (som >= 500 && som < 800)
      {
        noTone(BUZZER_1);
        tone(BUZZER_2, 2400);
      }
      else
      {
        desligarBuzzers();
      }
    }
    else
    {
      desligarBuzzers();
    }

    return;
  }


  // ==========================================================
  // CRÍTICO
  // ==========================================================

  if (estadoAtual == CRITICO)
  {
    desligarLEDs();

    unsigned long ciclo = agora % 500;

    if (ciclo < 250)
    {
      digitalWrite(LED_VERMELHO, HIGH);
    }
    else
    {
      digitalWrite(LED_VERMELHO, LOW);
    }


    if (!buzzerSilenciado)
    {
      unsigned long som = agora % 1000;

      if (som < 180)
      {
        tone(BUZZER_1, 2300);
        tone(BUZZER_2, 2700);
      }
      else if (som >= 300 && som < 480)
      {
        tone(BUZZER_1, 2300);
        tone(BUZZER_2, 2700);
      }
      else
      {
        desligarBuzzers();
      }
    }
    else
    {
      desligarBuzzers();
    }

    return;
  }
}


// ============================================================
// BOTÃO
// ============================================================

void atualizarBotao()
{
  bool leitura = digitalRead(BOTAO) == LOW;

  unsigned long agora = millis();


  // ==========================================================
  // AGUARDA O BOTÃO SER SOLTO APÓS O BOOT
  // ==========================================================

  if (!botaoInicializado)
  {
    if (!leitura)
    {
      botaoInicializado = true;
    }

    return;
  }


  // ==========================================================
  // DEBOUNCE
  // ==========================================================

  if (leitura != botaoPressionado)
  {
    if (agora - ultimaMudancaBotao < TEMPO_DEBOUNCE)
    {
      return;
    }

    ultimaMudancaBotao = agora;

    botaoPressionado = leitura;


    // ========================================================
    // BOTÃO FOI PRESSIONADO
    // ========================================================

    if (botaoPressionado)
    {
      inicioPressaoBotao = agora;

      return;
    }


    // ========================================================
    // BOTÃO FOI SOLTO
    // ========================================================

    unsigned long duracao =
      agora - inicioPressaoBotao;


    // --------------------------------------------------------
    // LONG PRESS
    // --------------------------------------------------------

    if (duracao >= TEMPO_LONGO)
    {
      Serial.println();
      Serial.println("==========================================");
      Serial.println("REINICIANDO SISTEMA...");
      Serial.println("==========================================");

      desligarBuzzers();
      desligarLEDs();

      delay(100);

      ESP.restart();

      return;
    }


    // --------------------------------------------------------
    // CURTO
    // --------------------------------------------------------

    if (duracao >= 80)
    {
      buzzerSilenciado = true;

      inicioSilenciamento = agora;

      desligarBuzzers();

      Serial.println();
      Serial.println("BUZZER SILENCIADO POR 60 SEGUNDOS.");
      Serial.println();
    }
  }


  // ==========================================================
  // NÃO ALTERA ESTADO DO ALAGAMENTO
  // ==========================================================
}


// ============================================================
// SERIAL
// ============================================================

void imprimirStatus()
{
  if (millis() - ultimoSerial < INTERVALO_SERIAL)
  {
    return;
  }

  ultimoSerial = millis();


  Serial.print("Distancia: ");
  Serial.print(distanciaAtual, 2);

  Serial.print(" cm | Agua: ");
  Serial.print(aguaAtual, 2);

  Serial.print(" cm | Subida: ");
  Serial.print(taxaSubida, 2);

  Serial.print(" cm/min | Estado: ");
  Serial.print(nomeEstado(estadoAtual));


  if (subidaRapida)
  {
    Serial.print(" | SUBIDA RAPIDA");
  }


  if (buzzerSilenciado)
  {
    unsigned long restante =
      TEMPO_SILENCIO - (millis() - inicioSilenciamento);

    unsigned long segundos =
      restante / 1000;

    Serial.print(" | BUZZER SILENCIADO: ");
    Serial.print(segundos);
    Serial.print("s");
  }


  if (!sensorOK)
  {
    Serial.print(" | SENSOR OFFLINE");
  }


  // ═══ NOVO ═══  Status de rede
  if (wifiConectado && WiFi.status() == WL_CONNECTED)
  {
    Serial.print(" | WiFi: ");
    Serial.print(WiFi.RSSI());
    Serial.print("dBm");

    Serial.print(" | Envios: ");
    Serial.print(totalEnvios);

    Serial.print(" | Falhas: ");
    Serial.print(totalFalhas);
  }
  else
  {
    Serial.print(" | WiFi: OFFLINE");
  }

  Serial.println();
}

// ============================================================
// SETUP
// ============================================================

void setup()
{
  Serial.begin(115200);

  delay(500);


  // ==========================================================
  // PINOS
  // ==========================================================

  pinMode(LED_VERMELHO, OUTPUT);
  pinMode(LED_AMARELO, OUTPUT);
  pinMode(LED_VERDE, OUTPUT);

  pinMode(BUZZER_1, OUTPUT);
  pinMode(BUZZER_2, OUTPUT);

  pinMode(BOTAO, INPUT_PULLUP);

  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);


  // ==========================================================
  // ESTADO INICIAL DOS PINOS
  // ==========================================================

  desligarLEDs();
  desligarBuzzers();

  digitalWrite(TRIG, LOW);


  // ==========================================================
  // CABEÇALHO
  // ==========================================================

  Serial.println();
  Serial.println("==========================================");
  Serial.println("      MONITOR DE ALAGAMENTO V5");
  Serial.println("==========================================");

  Serial.print("LED vermelho : GPIO ");
  Serial.println(LED_VERMELHO);

  Serial.print("LED amarelo  : GPIO ");
  Serial.println(LED_AMARELO);

  Serial.print("LED verde    : GPIO ");
  Serial.println(LED_VERDE);

  Serial.print("Buzzer 1     : GPIO ");
  Serial.println(BUZZER_1);

  Serial.print("Buzzer 2     : GPIO ");
  Serial.println(BUZZER_2);

  Serial.print("Botao        : GPIO ");
  Serial.println(BOTAO);

  Serial.print("TRIG         : GPIO ");
  Serial.println(TRIG);

  Serial.print("ECHO         : GPIO ");
  Serial.println(ECHO);

  Serial.println();

  Serial.print("Distancia piso configurada: ");
  Serial.print(DISTANCIA_PISO_CM, 2);
  Serial.println(" cm");

  Serial.println();


  // ==========================================================
  // TESTE
  // ==========================================================

  testeInicial();


  // ══════════════════════════════════════════════════════════
  // ═══ NOVO ═══  WI-FI
  // ══════════════════════════════════════════════════════════

  conectarWiFi();


  // ==========================================================
  // SENSOR INICIAL
  // ==========================================================

  float primeiraLeitura = medirDistancia();

  if (primeiraLeitura > 0)
  {
    sensorOK = true;

    falhasSensor = 0;
    recuperacoesSensor = RECUPERACOES_SENSOR;

    distanciaFiltrada = filtrarDistancia(primeiraLeitura);

    distanciaAtual = distanciaFiltrada;

    aguaAtual = calcularAgua(distanciaAtual);

    aguaAnterior = aguaAtual;

    ultimaTaxa = millis();

    taxaSubida = 0.0;
    subidaRapida = false;
  }
  else
  {
    sensorOK = false;

    Serial.println("ATENCAO: sensor sem leitura inicial.");
  }


  // ==========================================================
  // ESTADO INICIAL
  // ==========================================================

  if (sensorOK)
  {
    estadoAtual =
      determinarEstadoDireto(aguaAtual);

    estadoCandidato = estadoAtual;

    Serial.println();
    Serial.println("==========================================");
    Serial.print("ESTADO INICIAL: ");
    Serial.println(nomeEstado(estadoAtual));
    Serial.println("==========================================");
  }
  else
  {
    estadoAtual = SENSOR_FALHA;
    estadoCandidato = SENSOR_FALHA;
  }


  // ==========================================================
  // BOTÃO
  // ==========================================================

  // Não habilita imediatamente.
  // O usuário precisa soltar o botão depois do boot.
  botaoInicializado = false;
  botaoPressionado = false;


  // ==========================================================
  // INÍCIO
  // ==========================================================

  Serial.println();
  Serial.println("Sistema iniciado.");
  Serial.println("Botao curto = silencia buzzer por 60s.");
  Serial.println("Botao 2s = reinicia ESP32.");
  Serial.println();
}


// ============================================================
// LOOP
// ============================================================

void loop()
{
  unsigned long agora = millis();


  // ==========================================================
  // SENSOR
  // ==========================================================

  if (agora - ultimaLeituraSensor >= INTERVALO_SENSOR)
  {
    ultimaLeituraSensor = agora;

    atualizarSensor();

    atualizarEstado();
  }


  // ==========================================================
  // BOTÃO
  // ==========================================================

  atualizarBotao();


  // ==========================================================
  // ALARME
  // ==========================================================

  atualizarAlarme();


  // ==========================================================
  // SERIAL
  // ==========================================================

  imprimirStatus();


  // ══════════════════════════════════════════════════════════
  // ═══ NOVO ═══  ENVIO PARA A NUVEM
  // ══════════════════════════════════════════════════════════
  //
  // Envia a cada 5s OU imediatamente se o estado mudou.
  //
  // NUNCA bloqueia alarme local: LEDs e buzzers rodam
  // independentemente em atualizarAlarme().
  // ══════════════════════════════════════════════════════════

  bool estadoMudou = ((int)estadoAtual != ultimoEstadoEnviado);
  bool horaDeEnviar = (agora - ultimoEnvio >= INTERVALO_ENVIO);

  if (estadoMudou || horaDeEnviar)
  {
    // Evita spam em mudanças rápidas (mínimo 500ms entre envios)
    if (agora - ultimoEnvio > 500)
    {
      enviarTelemetria();

      ultimoEnvio = agora;
      ultimoEstadoEnviado = (int)estadoAtual;
    }
  }
}
